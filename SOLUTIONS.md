```python
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
```

### cousins-in-binary-tree

_source bank: SOLUTIONS_MASTER.json_

```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def isCousins(self, root: Optional[TreeNode], x: int, y: int) -> bool:
        info = {}
        def dfs(n, parent, depth):
            if not n: return
            if n.val in (x, y): info[n.val] = (parent, depth)
            dfs(n.left, n.val, depth + 1)
            dfs(n.right, n.val, depth + 1)
        dfs(root, None, 0)
        px, dx = info[x]; py, dy = info[y]
        return px != py and dx == dy
```

```python
        return ans
```

```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def isUnivalTree(self, root: Optional[TreeNode]) -> bool:
        def dfs(n, v):
            if not n: return True
            return n.val == v and dfs(n.left, v) and dfs(n.right, v)
        return dfs(root, root.val)
```

```python
class Solution:
    def validPalindrome(self, s: str) -> bool:
        def is_pal(lo, hi):
            while lo < hi:
                if s[lo] != s[hi]: return False
                lo += 1; hi -= 1
            return True
        i, j = 0, len(s) - 1
        while i < j:
            if s[i] != s[j]:
                return is_pal(i+1, j) or is_pal(i, j-1)
            i += 1; j -= 1
        return True
```

```python
        rank = {c: i for i, c in enumerate(order)}
```

```python
class Solution:
    def wordPattern(self, pattern: str, s: str) -> bool:
        words = s.split()
        if len(pattern) != len(words): return False
        m1, m2 = {}, {}
        for p, w in zip(pattern, words):
            if p in m1 and m1[p] != w: return False
            if w in m2 and m2[w] != p: return False
            m1[p] = w; m2[w] = p
        return True
```

```python
class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        R, C = len(board), len(board[0])
        def dfs(i, j, k):
            if k == len(word): return True
            if i < 0 or j < 0 or i >= R or j >= C or board[i][j] != word[k]: return False
            tmp = board[i][j]
            board[i][j] = '#'
            for di, dj in ((1,0),(-1,0),(0,1),(0,-1)):
                if dfs(i+di, j+dj, k+1): return True
            board[i][j] = tmp
            return False
        for i in range(R):
            for j in range(C):
                if dfs(i, j, 0): return True
        return False
```
